# Job-runner library dependencies.
#
# Pinned to compatible-release ranges so a patch fix is picked up on rebuild
# while a breaking major cannot be. Everything else the library needs is in the
# Python standard library.
#
# uvicorn is installed without the [standard] extras: those add uvloop,
# httptools, watchfiles, and websockets, none of which this API uses, and each
# is a native wheel that has to exist for the image's platform.
fastapi~=0.115
uvicorn~=0.30

# Required by fastapi.testclient, for running the library's own test suite.
# Not needed at runtime.
httpx~=0.27
